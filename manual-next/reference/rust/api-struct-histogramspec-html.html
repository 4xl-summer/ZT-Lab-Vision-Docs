new

Creates a specification with 2 through 65,536 bins and finite, strictly increasing representable edges. The specifications (256, 0, 256) and (65536, 0, 65536) have exact integer edges. Validation takes O(bin_count) time. use zvision_core::ZImage; use zvision_ops::{HistogramSpec, histogram_image, otsu_threshold_image}; let image = ZImage::from_vec_gray16(4, 1, vec![0, 0, 60_000, 60_000]).unwrap(); let spec = HistogramSpec::new(65_536, 0.0, 65_536.0).unwrap(); let histogram = histogram_image(&image, spec).unwrap(); assert_eq!(histogram.channels()[0][60_000], 2); assert_eq!(otsu_threshold_image(&image, spec).unwrap(), 1.0);
