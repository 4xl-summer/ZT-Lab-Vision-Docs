전체 호출 예제 · C# 검사 API

C# 검사 API에서 전체 호출 예제을 이해하기 위한 입력, 동작과 결과 설명

상위 목차로 돌아가기

전체 호출 예제

C#
using System;
using ZVision;

class Program {
    static void Main() {
        byte[] motif = { 1,2,7, 9,5,3, 4,8,6 };
        byte[] referencePixels = new byte[25];
        for (int y=0; y<3; y++) for (int x=0; x<3; x++)
            referencePixels[(y+1)*5+x+1] = motif[y*3+x];
        using var reference = ZImage.FromGray8(5, 5, referencePixels);
        var search = new MatchSpec(0, 0, 1, 1, 1, 1, .999999, 0, 1);
        var alignmentSpec = new AlignmentSpec(new InspectionRect(1,1,3,3), search, Interpolation.Nearest, 1);
        foreach (string name in new[]{"normal", "moved", "lighting", "spot", "missing"}) {
            byte[] pixels = new byte[49];
            int shift = name == "normal" ? 1 : 2;
            if (name != "missing")
                for (int y=0; y<3; y++) for (int x=0; x<3; x++)
                    pixels[(y+shift)*7+x+shift] = motif[y*3+x];
            if (name == "lighting")
                for (int i=0; i<pixels.Length; i++) pixels[i] += 20;
            if (name == "spot") pixels[1*7+1] = 100;
            using var source = ZImage.FromGray8(7, 7, pixels);
            using var alignment = source.AlignReference(reference, alignmentSpec);
            if (!alignment.Found) {
                Console.WriteLine($"{name}: alignment unavailable");
                continue;
            }
            using var aligned = alignment.Aligned;
            using var valid = alignment.ValidMask;
            using var difference = aligned.InspectionDifference(reference, valid,
                new DifferenceSpec(10, PhotometricCorrection.Offset));
            using var mask = difference.Mask;
            using var score = difference.Score;
            using var defects = mask.SurfaceDefects(score, new DefectSpec(8,1,8,2,4));
            using var decision = defects.Judge(new DecisionSpec(0,0,0, new InspectionRect(0,0,5,5)));
            Console.WriteLine($"{name}: valid={alignment.Info.ValidCount}, count={decision.Info.DefectCount}, pass={decision.Passed}, reasons={decision.Info.Reasons}");
        }
    }
}

이 호출의 핵심은 기준과 검사 입력을 먼저 준비한 다음 정렬 결과의 성공 여부를 검사하는 것입니다. 성공했을 때만 정렬 영상과 유효 마스크를 꺼내어 같은 기준 격자에서 차분합니다. 차분 결과의 점수와 이진 마스크는 서로 다른 영상이며, 연결 요소 추출에는 두 영상을 함께 전달합니다. 추출된 후보를 지정한 개수와 면적 및 허용 영역으로 평가하면 최종 판정과 이유를 읽을 수 있습니다.

정상과 결함 입력을 비교할 때에는 정렬이 실제로 발견되었는지, 유효 마스크가 기대한 영역을 선택했는지부터 확인합니다. 밝기나 결함 점수만 보고 정렬 상태를 추측하지 않습니다. 코드에서 획득한 이미지와 중간 결과를 각각 해제하는 순서가 명시된 이유는 각 반환값이 별도 수명을 갖기 때문입니다. 이 예제의 수치는 동작을 확인하기 위한 합성 값입니다.

실제로 사용할 때는 각 중간 결과를 단계별로 저장하거나 살펴보면서 마스크가 어떤 픽셀을 선택했는지 확인하는 것이 판정 원인을 이해하는 데 도움이 됩니다.

정렬 결과가 미발견이면 차분 영상을 요청하는 대신 해당 입력을 검사 불가로 기록합니다. 성공 사례에서는 유효 마스크와 기준이 같은 원점을 갖는지 살펴보고 후보 수와 판정 이유를 함께 비교합니다. 이렇게 하면 위치 실패와 실제 결함을 서로 다른 원인으로 조사할 수 있습니다.
